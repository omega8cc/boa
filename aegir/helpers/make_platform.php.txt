<?php

  // Idempotency guard: re-running the installer must not create duplicate
  // platform (and profile package) nodes. Derive the context name exactly the
  // way hosting_platform_insert() does and no-op when it is already
  // registered. PHP 5.6-safe.
  if (function_exists('_hosting_platform_context_name')) {
    $context = 'platform_' . _hosting_platform_context_name($_SERVER['argv'][3]);
  }
  else {
    $context = 'platform_' . preg_replace('/[^\w.-]/', '', $_SERVER['argv'][3]);
  }
  if (function_exists('hosting_context_load') && hosting_context_load($context)) {
    drush_log(dt('make_platform: context !c already registered, skipping.', array('!c' => $context)), 'ok');
    return;
  }

  // Create the platform node
  global $argv;
  var_dump($argv);
  $types = node_types_rebuild();
  $node = new stdClass();
  $node->type = 'platform';
  $node->title = $_SERVER['argv'][3];
  $node->language = LANGUAGE_NONE;
  node_object_prepare($node);
  $node->uid = 1;
  $node->publish_path = $_SERVER['argv'][5];
  $node->web_server = variable_get('hosting_default_web_server', 2);
  $node->status = 1;
  $node = node_submit($node);
  node_save($node);
  $platform_id = $node->nid;
  variable_set('hosting_own_platform', $node->nid);

  // Create the platform profile node
  $node = new stdClass();
  $node->type = 'package';
  $node->title = $_SERVER['argv'][3];
  $node->language = LANGUAGE_NONE;
  node_object_prepare($node);
  $node->uid = 1;
  $node->package_type = 'profile';
  $node->short_name = $_SERVER['argv'][4];
  $node->status = 1;
  $node = node_submit($node);
  node_save($node);

?>
